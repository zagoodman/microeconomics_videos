% Automated endnotes (placed on a new page titled NOTES before Acknowledgments/References)
\newcounter{jeeendnote}
\newwrite\@endnotes
\newif\if@endnotesopen
\@endnotesopenfalse
\newcommand{\jeeendnote}[1]{%
	\unskip
	\stepcounter{jeeendnote}%
	\textsuperscript{\thejeeendnote}%
	\if@endnotesopen\else
		\immediate\openout\@endnotes=\jobname.ent\relax
		\@endnotesopentrue
	\fi
	\immediate\write\@endnotes{\string\jeeendnoteitem{\thejeeendnote}{\unexpanded{#1}}}%
}
\newcommand{\jeeendnoteitem}[2]{%
	\par\noindent\textsuperscript{#1}\hspace{0.5em}#2\par
}
\newcommand{\theendnotes}{%
	\if@endnotesopen
		\immediate\closeout\@endnotes
		\@endnotesopenfalse
		\clearpage
		\section*{NOTES}
		\begingroup
		\doublespacing
		\InputIfFileExists{\jobname.ent}{}{}%
		\endgroup
	\fi
}
\let\footnote=\jeeendnote
